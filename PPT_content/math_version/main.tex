% ============================================================
% SLIDE 2 — Problem at a glance
% ============================================================
\begin{frame}{Problem at a Glance}
\begin{center}
\large
$N$ \textbf{drones} fly over $M$ \textbf{ground sensors}.\\[0.4em]
Each drone has a \textbf{finite battery} and can \textbf{fail}.\\[0.4em]
Every sensor needs \textbf{at least $K$ drones} above it \textbf{at all times}.
\end{center}

\vspace{0.6em}
\textbf{Challenge:} coordinate the swarm with \emph{no central controller}.\\
\textbf{Solution:} each drone learns its own policy via multi-agent RL.
\end{frame}

% ============================================================
% SLIDE 3 — MDP Formulation
% ============================================================
\begin{frame}{MDP Formulation}
We model the system as a cooperative Dec-POMDP:
\[
\mathcal{G} = \bigl\langle\,
  \underbrace{N}_{\text{agents}},\;
  \underbrace{\mathcal{S}}_{\text{state}},\;
  \underbrace{\{\mathcal{A}_i\}}_{\text{actions}},\;
  \underbrace{\{\mathcal{O}_i\}}_{\text{obs.}},\;
  \underbrace{\mathcal{T}}_{\text{dynamics}},\;
  \underbrace{R}_{\text{reward}},\;
  \underbrace{\gamma}_{\text{discount}}
\,\bigr\rangle
\]

\vspace{0.3em}
\textbf{Objective:} find a policy $\pi$ maximising expected return:
\[
\max_{\pi}\;
\mathbb{E}_{\pi}
\!\left[
  \sum_{t=0}^{T} \gamma^{t}\, R_t
\right]
\]

\vspace{0.3em}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Symbol} & \textbf{Meaning} \\
\midrule
$N$ & number of active UAV agents \\
$\mathcal{S}$ & global system state space \\
$\mathcal{A}_i$ & action space per UAV \\
$\mathcal{O}_i$ & local observation per UAV \\
$\mathcal{T}$ & state transition dynamics \\
$R$ & cooperative reward function \\
$\gamma \in [0,1)$ & discount factor \\
\bottomrule
\end{tabular}
\end{frame}

% ============================================================
% SLIDE 4 — Global State
% ============================================================
\begin{frame}{Global State $s_t$}
The full system snapshot at time $t$:
\[
s_t = \Bigl\{
  (p_i,\; v_i,\; b_i,\; m_i)_{i=1}^{N},\;\;
  (q_d)_{d=1}^{M},\;\;
  Q_t,\;\;
  F_t
\Bigr\}
\]

\vspace{0.4em}
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{Symbol} & \textbf{Meaning} & \textbf{Range} \\
\midrule
$p_i$ & position of UAV $i$ & $\mathbb{R}^3$ \\
$v_i$ & velocity of UAV $i$ & $\|v_i\| \le v_{\max}$ \\
$b_i$ & normalised battery level & $[0,1]$ \\
$m_i$ & operating mode & \{serve, transit, charge\} \\
$q_d$ & position of sensor $d$ & fixed \\
$Q_t$ & charging pad occupancy & $\{0,1\}^S$ \\
$F_t$ & which UAVs have failed & $\{0,1\}^N$ \\
\bottomrule
\end{tabular}

\vspace{0.3em}
\begin{block}{Key}
The \textbf{centralised critic} sees all of $s_t$ during \textbf{training}.
Each UAV only sees its \textbf{local observation} during \textbf{execution}.
\end{block}
\end{frame}

% ============================================================
% SLIDE 5 — Local Observation
% ============================================================
\begin{frame}{Local Observation $o_i$}
Each UAV sees only what is nearby:
\[
o_i = \bigl[
  \underbrace{p_i,\; v_i,\; b_i}_{\text{own state}},\;\;
  \underbrace{\mathcal{N}_i}_{\text{neighbours}},\;\;
  \underbrace{C_i^{\text{local}}}_{\text{local coverage}},\;\;
  \underbrace{\text{risk}_i}_{\text{cut-vertex?}},\;\;
  \underbrace{t_i^{\text{empty}}}_{\text{time-to-empty}},\;\;
  \underbrace{E_i^{\text{station}}}_{\text{energy to pad}}
\bigr]
\]

\vspace{0.4em}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Symbol} & \textbf{Meaning} \\
\midrule
$\mathcal{N}_i$ & indices of UAVs within $R_{\text{comm}}$ \\
$C_i^{\text{local}}$ & number of local sensors covered \\
$\text{risk}_i$ & $=1$ if removing $i$ disconnects $G_t$ \\
$t_i^{\text{empty}}$ & seconds until battery $\le$ return threshold \\
$E_i^{\text{station}}$ & energy needed to reach nearest pad \\
\bottomrule
\end{tabular}

\vspace{0.3em}
\begin{alertblock}{Decentralised execution}
At deployment: $a_i \sim \pi_\theta(a_i \mid o_i)$ --- no global state needed.
\end{alertblock}
\end{frame}

% ============================================================
% SLIDE 6 — Why Decentralised?
% ============================================================
\begin{frame}{Why Not a Central Controller?}
\textbf{Centralised} action space grows exponentially:
\[
|\mathcal{A}| = A^N
\]

\vspace{0.3em}
\begin{center}
\begin{tabular}{@{}ccccc@{}}
\toprule
$N$ & 4 & 6 & 10 & 20 \\
\midrule
$|\mathcal{A}|$ & $5^4 = 625$ & $5^6 = 15625$ & $\sim\!10^7$ & $\sim\!10^{14}$ \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.4em}
\textbf{Solution:} factorised policy --- each UAV acts independently:
\[
a_i \sim \pi_\theta(a_i \mid o_i)
\qquad\Longrightarrow\qquad
|\mathcal{A}|_{\text{eff}} = A \;\text{ per agent}
\]

\vspace{0.3em}
\begin{block}{Trade-off}
Decentralised is tractable, but each UAV must infer coordination from its local observation $o_i$ alone.
\end{block}
\end{frame}

% ============================================================
% SLIDE 7 — Action Space
% ============================================================
\begin{frame}{Action Space $\mathcal{A}_i$}
Each UAV picks one action per time step:
\[
\mathcal{A}_i = \bigl\{
  \text{Hover},\;
  {+x},\; {-x},\; {+y},\; {-y},\;
  \text{Charge}
\bigr\}, \quad |\mathcal{A}_i| = 6
\]

\vspace{0.5em}
\textbf{Movement:}
\[
p_i(t+1) = p_i(t) + v_i(t)\,\Delta t,
\quad \|v_i\| \le v_{\max},\;\; p_i \in \Omega
\]

\vspace{0.3em}
\textbf{Charge is valid only when:}
\[
\|p_i - p_s^{\text{pad}}\| \le d_{\text{charge}}
\quad\text{and}\quad
Q_t[s] = 1 \;\text{(pad is free)}
\]
Otherwise the action defaults to \emph{Hover}.
\end{frame}

% ============================================================
% SLIDE 8 — K-Coverage
% ============================================================
\begin{frame}{K-Coverage Requirement}
\textbf{Rule:} every sensor $d$ must have $\ge K$ drones above it.

\vspace{0.4em}
\textbf{Binary coverage} (is UAV $i$ covering sensor $d$?):
\[
x_{id}(t) = \mathbf{1}\!\bigl[\,\|p_i(t) - q_d\| \le R_{\text{cov}}\,\bigr]
\]

\vspace{0.3em}
\textbf{Count per sensor:}
\[
n_d(t) = \sum_{i=1}^{N} x_{id}(t)
\]

\vspace{0.3em}
\textbf{Coverage score:}
\[
C(t) = \frac{1}{M}\sum_{d=1}^{M} \min\!\left(\frac{n_d(t)}{K},\; 1\right)
\]

\vspace{0.3em}
\textbf{Under-coverage rate} (to minimise):
\[
U(t) = \frac{1}{M}\sum_{d=1}^{M} \mathbf{1}\!\bigl[\,n_d(t) < K\,\bigr]
\]
\end{frame}

% ============================================================
% SLIDE 9 — Connectivity
% ============================================================
\begin{frame}{Connectivity}
Drones form a \textbf{communication graph}:
\[
G_t = (V, E_t),
\quad
E_t = \bigl\{(i,j) : \|p_i - p_j\| \le R_{\text{comm}}\bigr\}
\]

\vspace{0.5em}
\textbf{Connectivity risk} (is UAV $i$ a cut-vertex?):
\[
\text{risk}_i(t) = \mathbf{1}\!\bigl[\,G_t^{-i} \text{ is disconnected}\,\bigr]
\]

\vspace{0.5em}
\begin{block}{Why this matters}
A drone that is the \emph{only link} between two parts of the swarm
should \textbf{not} leave for charging without a replacement.
\end{block}

\vspace{0.3em}
\textbf{Desired property:}
$G_t$ is connected $\;\forall\, t$.
\end{frame}

% ============================================================
% SLIDE 10 — Battery Model
% ============================================================
\begin{frame}{Battery Model}
Three sources of energy drain:
\[
b_i(t+\Delta t) = b_i(t)
\underbrace{- \frac{P_{\text{hover}} + P_{\text{move}} + P_{\text{comm}}}{B_i^{\max}}\,\Delta t}_{\text{drain}}
+ \underbrace{\Delta b_i^{\text{charge}}}_{\text{recharge (at pad)}}
\]

\vspace{0.5em}
\textbf{Return-to-charge condition:}
\[
b_i(t) \cdot B_i^{\max}
\;\le\;
E_i^{\text{req}}
= E_i^{\text{station}} + E_i^{\text{return}} + E^{\text{reserve}}
\]

\vspace{0.3em}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Symbol} & \textbf{Meaning} \\
\midrule
$P_{\text{hover}}$ & power consumed hovering (W) \\
$P_{\text{move}}$ & power consumed moving (W) \\
$P_{\text{comm}}$ & power consumed communicating (W) \\
$B_i^{\max}$ & total battery capacity (J) \\
$E_i^{\text{station}}$ & energy to reach nearest pad \\
$E_i^{\text{return}}$ & energy to return from pad \\
$E^{\text{reserve}}$ & safety margin \\
\bottomrule
\end{tabular}
\end{frame}

% ============================================================
% SLIDE 11 — Charging Coordination
% ============================================================
\begin{frame}{Charging Coordination}
Multiple drones may need to charge, but pads are limited.
We solve a \textbf{min-cost assignment}.

\vspace{0.3em}
\textbf{Decision:} $y_{is} \in \{0,1\}$ --- assign UAV $i$ to pad $s$?

\vspace{0.3em}
\textbf{Cost:}
\[
c_{is} = \alpha\, d_{is} + \beta\, q_s + \mu\, u_i
\]

\vspace{0.3em}
\textbf{Optimisation:}
\[
\min_{y}\; \sum_{i}\sum_{s}\; c_{is}\, y_{is}
\]

\vspace{0.3em}
\textbf{Constraints:}
\[
\sum_{s} y_{is} \le 1 \;\;\forall\, i
\qquad(\text{each UAV} \le \text{1 pad}),
\qquad
\sum_{i} y_{is} \le 1 \;\;\forall\, s
\qquad(\text{each pad} \le \text{1 UAV})
\]

\vspace{0.3em}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Symbol} & \textbf{Meaning} \\
\midrule
$d_{is}$ & distance from UAV $i$ to pad $s$ \\
$q_s$ & queue length at pad $s$ \\
$u_i$ & urgency (battery depletion risk) \\
$\alpha, \beta, \mu$ & weights \\
\bottomrule
\end{tabular}

Solved efficiently via the \textbf{Hungarian algorithm}.
\end{frame}

% ============================================================
% SLIDE 12 — Objective Function
% ============================================================
\begin{frame}{Multi-Objective Function}
\[
\min_{\pi}\;
\mathbb{E}_{\pi}
\!\left[
\sum_{t=0}^{T-1}
\Bigl(
  \underbrace{\lambda_O\, U(t)}_{\text{under-coverage}}
  + \underbrace{\lambda_E\, E(t)}_{\text{energy waste}}
  + \underbrace{\lambda_D\, D(t)}_{\text{dead time}}
  + \underbrace{\lambda_W\, W(t)}_{\text{wasted moves}}
  + \underbrace{\lambda_R\, R(t)}_{\text{recovery}}
  + \underbrace{\lambda_C\, \text{Crit}(t)}_{\text{cut-vertex}}
\Bigr)
\right]
\]

\vspace{0.4em}
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{Term} & \textbf{Definition} & \textbf{Penalises} \\
\midrule
$U(t)$ & $\frac{1}{M}\sum_d \mathbf{1}[n_d{<}K]$ & sensors below $K$-coverage \\
$E(t)$ & fraction of flights w/ low battery & excessive energy drain \\
$D(t)$ & fraction of time degraded & time after a failure \\
$W(t)$ & moves that don't help coverage & wasteful repositioning \\
$R(t)$ & $\min(t_{\text{recovered}}, t_{\text{cap}})$ & time to restore coverage \\
$\text{Crit}(t)$ & fraction of active cut-vertices & connectivity fragility \\
\bottomrule
\end{tabular}
\end{frame}

% ============================================================
% SLIDE 13 — Constraints
% ============================================================
\begin{frame}{Safety Constraints}
At every time step $t$, for all UAVs $i$:

\vspace{0.5em}
\begin{enumerate}
    \item \textbf{Speed:}
    $\|v_i(t)\| \le v_{\max}$
    
    \vspace{0.3em}
    \item \textbf{Workspace:}
    $p_i(t) \in \Omega \subset \mathbb{R}^3$
    
    \vspace{0.3em}
    \item \textbf{Collision avoidance:}
    $\|p_i(t) - p_j(t)\| \ge d_{\text{safe}}
    \quad \forall\, j \ne i$
\end{enumerate}

\vspace{0.5em}
\begin{block}{Hard vs Soft}
These are \textbf{hard constraints}. Violated states are prevented (action masking) or heavily penalised in the reward.
\end{block}
\end{frame}

% ============================================================
% SLIDE 14 — Failure and Recovery
% ============================================================
\begin{frame}{Failure \& Recovery}
\textbf{Recovery time:} how long until coverage is restored after a failure at $t_f$:
\[
\tau_f = \inf\!\bigl\{\tau \ge 0 :
  n_d(t_f + \tau) \ge K \;\;\forall\, d \in \mathcal{M}\bigr\}
\]

\vspace{0.4em}
\textbf{Requirement:}
\[
\tau_f \le \tau_{\max}
\]

\vspace{0.5em}
\begin{block}{Reserve UAV Strategy}
Keep 1 drone in reserve with $b_i > b_{\text{threshold}}$.\\[0.2em]
When a drone fails, the reserve \textbf{immediately repositions} to fill the gap, reducing $\tau_f$.
\end{block}

\vspace{0.3em}
\textbf{Activation:} reserve takes off when $F_t$ changes (new failure detected).
\end{frame}

% ============================================================
% SLIDE 15 — MAPPO
% ============================================================
\begin{frame}{Learning Algorithm: MAPPO}
\textbf{Multi-Agent Proximal Policy Optimisation.}

\vspace{0.3em}
\textbf{Actor} (per UAV):
$a_i \sim \pi_\theta(a_i \mid o_i)$

\vspace{0.3em}
\textbf{Critic} (centralised, training only):
$V_\phi(s_t)$

\vspace{0.3em}
\textbf{Actor loss} (PPO clipped):
\[
\mathcal{L}^{\text{actor}}(\theta)
= -\mathbb{E}_t\!\left[
  \min\!\left(
    r_t(\theta)\,\hat{A}_t,\;
    \text{clip}\bigl(r_t(\theta), 1{-}\epsilon, 1{+}\epsilon\bigr)\,\hat{A}_t
  \right)
\right]
\]
where $r_t(\theta) = \frac{\pi_\theta(a_t \mid o_t)}{\pi_{\theta_{\text{old}}}(a_t \mid o_t)}$,
$\hat{A}_t = R_t - V_\phi(s_t)$ (GAE).

\vspace{0.3em}
\textbf{Critic loss:}
\[
\mathcal{L}^{\text{critic}}(\phi)
= \mathbb{E}_t\!\left[\bigl(R_t - V_\phi(s_t)\bigr)^2\right]
\]

\vspace{0.3em}
\textbf{Training:} centralised critic sees $s_t$. \textbf{Execution:} each UAV acts on $o_i$ alone.
\end{frame}
